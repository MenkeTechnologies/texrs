% The LaTeX layer texrs can honour, written in TeX.
%
% LaTeX is itself a program in TeX, so the faithful way to carry it is to define
% these AS TeX macros. What is here is the part that lives in the mouth and the
% expander; anything needing the stomach -- boxes, glue, fonts, output routines
% -- is absent because texrs has no stomach.
%
% A macro that draws something becomes a macro that yields its text, and a macro
% that sets state the stomach would read becomes one that consumes its arguments
% and produces nothing. That is the honest reading of "run this document" for an
% engine with a mouth and an expander: what the text says, minus what the
% packages would have drawn.
\catcode`\{=1 \catcode`\}=2 \catcode`\#=6 \catcode`\[=12 \catcode`\]=12
\catcode`\@=11
%
% ---- \@ifnextchar: the peek every optional argument rests on -------------
% latex.ltx builds this with \futurelet. Note the shape: it does NOT branch
% straight into the arm. It \lets a macro to the arm inside the conditional and
% calls it AFTER the \fi, because an arm expanded while the conditional is still
% open leaves the \fi to be found by whatever the arm reads next -- written the
% other way it swallows the \fi and every optional argument breaks.
\def\@ifnextchar#1#2#3{\let\@wantc=#1\def\@yes{#2}\def\@no{#3}%
  \futurelet\@nextc\@ifnch}
\def\@ifnch{\ifx\@nextc\@wantc \let\@arm=\@yes \else \let\@arm=\@no \fi \@arm}
% The starred forms rest on the same peek. With the star there,
% `\@firstoftwo{#1}' takes it as its own second argument and drops it --
% latex.ltx's trick, kept rather than reinvented. The star follows a control
% WORD, so the mouth has already skipped any space and the space-token branch
% latex.ltx carries is not needed here.
\def\@firstoftwo#1#2{#1}
\def\@secondoftwo#1#2{#2}
\def\@ifstar#1#2{\@ifnextchar *{\@firstoftwo{#1}}{#2}}
% `\long' and `\outer' restrict what a following \def accepts. Nothing here
% reads either restriction, so they expand to nothing and the \def runs.
\def\long{}
\def\outer{}
