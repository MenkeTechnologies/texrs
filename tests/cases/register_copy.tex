% value is only known when the program runs. \dimen1=\skip0 takes the glue's
% natural width, which is tex.web §430's coercion.

\dimen5=\skip0

\advance\dimen6 by \dimen5

\skipdef\skipZ=2 \skipZ=\skip0
\message{[\the\skip0][\the\skip1]}
\message{[\the\dimen5][\the\dimen6][\the\dimenZ]}
\message{[\the\skipZ]}
