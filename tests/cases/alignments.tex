% widths from the widest entry, \tabskip from the preamble, \noalign,
% \omit, \span and plain.tex's \multispan, periodic preambles (&&) from
% the first column and from a later one, plain.tex's \matrix, \pmatrix,
% \cases, the cedilla built with \ooalign, a \valign, and \settabs with \+.

\setbox0\vbox{\halign{#&#&#\cr a\span b&c\cr \multispan2 long entry&c\cr}}\showbox0
\setbox0\vbox{\halign{&[#]\cr a&b&c&d\cr}}\showbox0
\setbox0\vbox{\halign{x#&&y#&z#\cr a&b&c&d&e\cr}}\showbox0
\setbox0\hbox{$\matrix{a&b\cr c&d\cr}$}\showbox0
\setbox0\hbox{$\pmatrix{1&0\cr 0&1\cr}$}\showbox0
