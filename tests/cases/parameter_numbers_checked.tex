% tex.web S476 and S479: a parameter text numbers its parameters 1 to 9 in
% order. A wrong number is `! Parameters must be numbered consecutively.',
% put back to be read again (as text) while the parameter takes the right
% number; a tenth is `! You already have nine parameters.' and is dropped with
% the token after it. In the body, `#' before anything but `#' or a number up
% to the count is `! Illegal parameter number in definition of \a.', put back,
% and the `#' kept as a macro parameter character.
\catcode`\{=1 \catcode`\}=2 \catcode`\#=6
\def\a#2{}\message{[\meaning\a]}
\gdef\a#1#3{}\message{[\meaning\a]}
\def\a#1#x{[#1]}\message{[\meaning\a]}
\def\a#1#2#3#4#5#6#7#8#9#0{}\message{[\meaning\a]}
\def\a#1{#2#1}\message{[\meaning\a]}
\def\a#1{#\relax}\message{[\meaning\a]}
\def\a#1{##1#1}\message{[\meaning\a][\a q]}
\def\a{#1}\message{[\meaning\a]}
\end
